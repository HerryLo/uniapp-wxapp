# 二期：辅食、用药、妈妈记录、旅行 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在日常记录中加入辅食、用药，并新增妈妈记录、旅行记录两个分包页面，同时保证导出/恢复兼容旧备份。

**Architecture:** 辅食、用药复用现有按日记录结构与 `RecordForm`；妈妈记录、旅行各自拥有独立纯逻辑模块和分包页面；备份模块增加两个可选数组字段，`schemaVersion` 保持 1。

**Tech Stack:** uni-app 3 + Vue 3 `<script setup>` + Vitest，目标平台 `mp-weixin` 与 H5。

## Global Constraints

> **2026-10-25 对齐说明（优先于下文早期代码片段）：**
>
> - 所有宝宝页面已使用 `<BabyPage>` 包裹并提供左侧栏；妈妈页、旅行页必须使用 `<BabyPage active="mom">` 和 `<BabyPage active="travel">`。
> - 侧栏入口顺序调整为：首页、喝奶、睡眠、排便、疫苗、生长、手记、妈妈、旅行、设置，共 10 项；侧栏列表需支持小屏滚动。
> - 温馨提示只在宝宝首页展示，妈妈页、旅行页不再使用 `Disclaimer`。
> - 全局采用奶油糖果风：卡片标题带 emoji，按钮、输入框、chips 使用现有全局样式，不新增旧版圆角和配色。
> - 日常记录 Tab 使用 emoji 标签：🍼 喝奶 / 😴 睡眠 / 💧 排便 / 🍚 辅食 / 💊 用药；5 个 Tab 使用 24rpx 字号适配侧栏后的宽度。
> - 记录页不再通过 `type` 自动打开表单，侧栏进入只切换 Tab。
> - 文档较早的“首页功能列表”“Disclaimer”“160rpx 侧栏”等描述以本节为准。

- 数据只保存在本机，不申请定位、不发网络请求、不登录。
- 日期使用本地时间字符串，禁止 `toISOString()`。
- 用药只记录用户自定义名称与用户手填用量，不预置药品、剂量或频次；表单必须提示「请遵医嘱」。
- 辅食反应只记录观察值（喜欢/一般/拒绝/异常），不做诊断；选择「异常」时提示咨询医生。
- 新增页面使用 `Disclaimer`，不放广告。
- 不新增依赖；所有新存储 key 以 `baby.` 开头。
- 纯逻辑任务先写失败测试再实现。
- 提交仅在用户授权后执行。

```js
function requiredTextError(value, label, max) {
  const text = String(value || '').trim()
  if (!text) return `请输入${label}`
  if (text.length > max) return `${label}最多 ${max} 个字`
  return ''
}

function optionalTextError(value, label, max) {
  const text = String(value || '').trim()
  if (text.length > max) return `${label}最多 ${max} 个字`
  return ''
}

export function validateFood(input) {
  if (!parseLocalDateTime(input.time)) return '请选择辅食时间'
  const nameError = requiredTextError(input.name, '食物名称', 30)
  if (nameError) return nameError
  const amountError = optionalTextError(input.amount, '份量', 20)
  if (amountError) return amountError
  if (!FOOD_REACTIONS.includes(input.reaction)) return '请选择进食反应'
  return optionalTextError(input.note, '备注', 200)
}

export function validateMedicine(input) {
  if (!parseLocalDateTime(input.time)) return '请选择用药时间'
  const nameError = requiredTextError(input.name, '药品名称', 30)
  if (nameError) return nameError
  const amountError = optionalTextError(input.amount, '用量', 20)
  if (amountError) return amountError
  return optionalTextError(input.note, '备注', 200)
}
```

```js
  { key: 'tired', emoji: '😪', label: '疲惫' },
  { key: 'anxious', emoji: '😟', label: '焦虑' },
  { key: 'low', emoji: '😔', label: '低落' },
]

const MOOD_KEYS = MOODS.map((item) => item.key)

function normalize(item) {
  if (!item || typeof item !== 'object' || Array.isArray(item)) return null
  if (!parseLocalDate(item.date)) return null
  if (!MOOD_KEYS.includes(item.mood)) return null
  const sleepHours = item.sleepHours == null ? null : Number(item.sleepHours)
  if (
    sleepHours != null &&
    (!Number.isFinite(sleepHours) || sleepHours < 0 || sleepHours > 24)
  ) {
    return null
  }
  return {
    date: item.date,
    sleepHours,
    mood: item.mood,
    condition: typeof item.condition === 'string' ? item.condition.slice(0, 50) : '',
    note: typeof item.note === 'string' ? item.note.slice(0, 200) : '',
    createdAt: Number.isFinite(item.createdAt) ? item.createdAt : 0,
    updatedAt: Number.isFinite(item.updatedAt) ? item.updatedAt : 0,
  }
}

export function loadMomEntries() {
  const raw = readStorage(MOM_KEY, [])
  if (!Array.isArray(raw)) return []
  const seen = new Set()
  return raw
    .map(normalize)
    .filter((item) => {
      if (!item || seen.has(item.date)) return false
      seen.add(item.date)
      return true
    })
    .sort((a, b) => b.date.localeCompare(a.date))
}

function validate(input) {
  if (!parseLocalDate(input.date)) return '请选择日期'
  if (isFutureDate(input.date)) return '日期不能晚于今天'
  if (!MOOD_KEYS.includes(input.mood)) return '请选择心情'
  if (input.sleepHours !== '' && input.sleepHours != null) {
    const sleepHours = Number(input.sleepHours)
    if (!Number.isFinite(sleepHours) || sleepHours < 0 || sleepHours > 24) {
      return '睡眠时长需在 0–24 小时'
    }
  }
  if (String(input.condition || '').trim().length > 50) {
    return '身体状态最多 50 个字'
  }
  if (String(input.note || '').trim().length > 200) {
    return '备注最多 200 个字'
  }
  return ''
}

export function saveMomEntry(input, now = Date.now()) {
  const error = validate(input)
  if (error) return { ok: false, error }
  const entries = loadMomEntries()
  const existing = entries.find((item) => item.date === input.date)
  const entry = {
    date: input.date,
    sleepHours:
      input.sleepHours === '' || input.sleepHours == null
        ? null
        : Number(input.sleepHours),
    mood: input.mood,
    condition: String(input.condition || '').trim(),
    note: String(input.note || '').trim(),
    createdAt: existing ? existing.createdAt : now,
    updatedAt: now,
  }
  const next = [entry, ...entries.filter((item) => item.date !== input.date)].sort(
    (a, b) => b.date.localeCompare(a.date)
  )
  if (!writeStorage(MOM_KEY, next)) return { ok: false, error: '保存失败' }
  return { ok: true, entry, entries: next }
}

export function removeMomEntry(date) {
  const entries = loadMomEntries()
  const next = entries.filter((item) => item.date !== date)
  if (next.length === entries.length) return { ok: true, entries }
  if (!writeStorage(MOM_KEY, next)) return { ok: false, error: '删除失败' }
  return { ok: true, entries: next }
}
```

```js
import { readStorage, writeStorage } from '../storage.js'
import { TRAVEL_KEY } from './keys.js'
import { parseLocalDate } from './date.js'

function makeId(now) {
  return `${now}-${Math.random().toString(36).slice(2, 8)}`
}

function normalize(item) {
  if (!item || typeof item !== 'object' || Array.isArray(item)) return null
  if (typeof item.id !== 'string') return null
  if (!parseLocalDate(item.startDate) || !parseLocalDate(item.endDate)) return null
  if (item.endDate < item.startDate) return null
  const place = typeof item.place === 'string' ? item.place.trim() : ''
  if (!place) return null
  return {
    id: item.id,
    startDate: item.startDate,
    endDate: item.endDate,
    place: place.slice(0, 30),
    note: typeof item.note === 'string' ? item.note.slice(0, 200) : '',
    createdAt: Number.isFinite(item.createdAt) ? item.createdAt : 0,
  }
}

export function loadTravels() {
  const raw = readStorage(TRAVEL_KEY, [])
  if (!Array.isArray(raw)) return []
  return raw
    .map(normalize)
    .filter(Boolean)
    .sort(
      (a, b) =>
        b.startDate.localeCompare(a.startDate) || b.createdAt - a.createdAt
    )
}

function validate(input) {
  if (!parseLocalDate(input.startDate)) return '请选择开始日期'
  if (!parseLocalDate(input.endDate)) return '请选择结束日期'
  if (input.endDate < input.startDate) return '结束日期不能早于开始日期'
  const place = String(input.place || '').trim()
  if (!place) return '请输入地点'
  if (place.length > 30) return '地点最多 30 个字'
  if (String(input.note || '').trim().length > 200) return '备注最多 200 个字'
  return ''
}

export function addTravel(input, now = Date.now()) {
  const error = validate(input)
  if (error) return { ok: false, error }
  const travel = {
    id: makeId(now),
    startDate: input.startDate,
    endDate: input.endDate,
    place: String(input.place).trim(),
    note: String(input.note || '').trim(),
    createdAt: now,
  }
  const list = [travel, ...loadTravels()].sort(
    (a, b) => b.startDate.localeCompare(a.startDate) || b.createdAt - a.createdAt
  )
  if (!writeStorage(TRAVEL_KEY, list)) return { ok: false, error: '保存失败' }
  return { ok: true, travel, list }
}

export function removeTravel(id) {
  const list = loadTravels()
  const next = list.filter((item) => item.id !== id)
  if (next.length === list.length) return { ok: true, list }
  if (!writeStorage(TRAVEL_KEY, next)) return { ok: false, error: '删除失败' }
  return { ok: true, list: next }
}
```
